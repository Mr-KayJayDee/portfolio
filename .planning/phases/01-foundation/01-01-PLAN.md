       '@nuxtjs/i18n',
       '@nuxt/eslint',
       '@nuxtjs/sitemap',
       'nuxt-gtag',
       '@nuxt/image'
     ],
     typescript: {
       strict: true
     },
     i18n: {
       locales: ['fr', 'en'],

       id: 'G-CDVVNFY6MV',

     }
   })
